binary_ops = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Pow: operator.pow,
    ast.Mod: operator.mod,
}

unary_ops = {
    ast.UAdd: operator.pos,
    ast.USub: operator.neg,
}

functions = {
    "sin": math.sin,
    "cos": math.cos,
    "tan": math.tan,
    "sqrt": math.sqrt,
    "log": math.log10,
    "ln": math.log,
    "factorial": math.factorial,
    "abs": abs,
    "exp": math.exp,
}

constants = {
    "pi": math.pi,
    "e": math.e,
}

def safe_eval(expression):
    tree = ast.parse(expression, mode="eval")

    def evaluate(node):
        if isinstance(node, ast.Constant):
            if type(node.value) in (int, float):
                return node.value

        if isinstance(node, ast.BinOp):
            op = binary_ops.get(type(node.op))
            if op:
                return op(evaluate(node.left), evaluate(node.right))

        if isinstance(node, ast.UnaryOp):
            op = unary_ops.get(type(node.op))
            if op:
                return op(evaluate(node.operand))

        if isinstance(node, ast.Name):
            if node.id in constants:
                return constants[node.id]

        if isinstance(node, ast.Call):
            if (isinstance(node.func, ast.Name)
                    and node.func.id in functions
                    and len(node.args) == 1
                    and not node.keywords):
                return functions[node.func.id](
                    evaluate(node.args[0])
                )

        raise ValueError("Invalid expression")

    result = evaluate(tree.body)

    if isinstance(result, complex):
        raise ValueError("Complex results not supported")

    if isinstance(result, float) and not math.isfinite(result):
        raise ValueError("Result out of range")

    return result




def press(value):
    display.insert(tk.END, value)

def clear():
    display.delete(0, tk.END)

def backspace():
    current = display.get()
    display.delete(0, tk.END)
    display.insert(0, current[:-1])

def calculate():
    try:
        expression = display.get()

        if not expression.strip():
            return

        result = safe_eval(expression)

        history.append(f"{expression} = {result}")

        display.delete(0, tk.END)
        display.insert(0, str(result))

        history_label.config(
            text="\n".join(history[-3:])
        )

    except Exception:
        display.delete(0, tk.END)
        display.insert(0, "Error")


def scientific(function):
    current = display.get()

    if current:
        display.delete(0, tk.END)
        display.insert(0, f"{function}({current})")
    else:
        display.insert(tk.END, function + "(")




buttons = [
    ["sin", "cos", "tan", "log"],
    ["ln", "sqrt", "pi", "e"],
    ["(", ")", "^", "DEL"],
    ["7", "8", "9", "/"],
    ["4", "5", "6", "*"],
    ["1", "2", "3", "-"],
    ["0", ".", "%", "+"],
    ["AC", "x²", "!", "="]
]

button_frame = tk.Frame(root, bg=BG)
button_frame.pack(padx=12, pady=18, fill="both", expand=True)

def button_action(value):
    if value == "AC":
        clear()

    elif value == "DEL":
        backspace()

    elif value == "=":
        calculate()

    elif value == "x²":
        press("**2")

    elif value == "!":
        scientific("factorial")

    elif value in ["sin", "cos", "tan", "log", "ln", "sqrt"]:
        scientific(value)

    elif value == "pi":
        press("pi")

    elif value == "e":
        press("e")

    else:
        press(value)


for r, row in enumerate(buttons):
    for c, value in enumerate(row):

        color = GREY
        text_color = WHITE

        if value in ["=", "+", "-", "*", "/"]:
            color = PURPLE

        if value == "=":
            color = GREEN
            text_color = "#10101c"

        if value in ["AC", "DEL"]:
            color = ORANGE
            text_color = "#10101c"

        if value in ["sin", "cos", "tan", "log",
                     "ln", "sqrt", "pi", "e"]:
            color = "#242b48"
            text_color = CYAN

        btn = tk.Button(
            button_frame,
            text=value,
            font=("Arial", 14, "bold"),
            bg=color,
            fg=text_color,
            activebackground=CYAN,
            activeforeground=BG,
            bd=0,
            relief="flat",
            cursor="hand2",
            command=lambda v=value: button_action(v)
        )

        btn.grid(
            row=r,
            column=c,
            padx=4,
            pady=5,
            sticky="nsew"
        )

for i in range(4):
    button_frame.columnconfigure(i, weight=1)

for i in range(8):
    button_frame.rowconfigure(i, weight=1)



def keyboard_input(event):

    key = event.keysym
    char = event.char

    if key in ("Return", "KP_Enter"):
        calculate()
        return "break"

    elif key == "BackSpace":
        backspace()
        return "break"

    elif key == "Escape":
        clear()
        return "break"

    elif char and char in "0123456789.+-*/()%":
        press(char)
        return "break"

    elif char == "^":
        press("**")
        return "break"

    return None

root.bind("<Key>", keyboard_input)
